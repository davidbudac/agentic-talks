A subagent is just another call

No special primitive — a fresh messages.create with its own message list.

def run_subagent(task):
  resp = client.messages.create(
    model=MODEL, max_tokens=400,
    system=SUBAGENT_SYSTEM, # shared, cached prefix
    messages=[{"role": "user", "content": task}], # all it knows
  )
  return resp.content[0].text # ← the single block that crosses back
